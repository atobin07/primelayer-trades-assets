import json
T='/root/.claude/projects/-home-claude-primelayer-trades-assets/ce86ac15-eb8a-50ab-85fe-970ee9b332f2.jsonl'
out={}
for line in open(T):
    try: r=json.loads(line)
    except: continue
    m=r.get('message',{})
    c=m.get('content') if isinstance(m,dict) else None
    if not isinstance(c,list): continue
    for b in c:
        if not isinstance(b,dict) or b.get('type')!='tool_result': continue
        cc=b.get('content')
        texts=[x.get('text','') for x in cc] if isinstance(cc,list) else [cc or '']
        for t in texts:
            if '"symbol"' in t and '"bars"' in t:
                try: d=json.loads(t); out[d["symbol"]]=d
                except Exception: pass
for k,d in out.items():
    print(k,d['count'],len(d['bars']['c']),d['summary']['first_open'],d['summary']['last_close'])
